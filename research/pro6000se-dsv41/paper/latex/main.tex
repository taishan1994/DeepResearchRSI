\documentclass[10pt,twocolumn]{article}

%% ---------- 中文支持 ----------
\usepackage[UTF8,fontset=none]{ctex}

%% 标点悬挂：中文双栏下唯一有效的「真溢出」解药。
%% 症状有辨识度：溢出行的末字符必是全角标点（」，。、）。
%% 原因不是可拉伸量不足，而是中文禁则不许标点居行首，
%% TeX 宁可越界也不拆——加多少 emergencystretch 都无用，因为缺的是断行点。
\usepackage{xeCJK}
\xeCJKsetup{PunctStyle=plain}

%% 字体：正文宋体、标题黑体、等宽 Consolas。
%% KaiTi 的族名是 KaiTi，不是 SimKai（写错会报 nullfont，且报错指不出是哪个字体）。
\setCJKmainfont{SimSun}[BoldFont=SimHei,ItalicFont=KaiTi]
%% SimHei 没有独立粗体字形，不声明会「粗体失效但只给警告」。
\setCJKsansfont{SimHei}[BoldFont=SimHei,AutoFakeBold=1.2]
\setCJKmonofont{SimHei}
\setmainfont{Times New Roman}
\setsansfont{Arial}
\setmonofont{Consolas}

%% 符号回退：md_to_tex.py 会把 → ⚠ ⭐ 这类 SimSun 没有的字形包进 \sym{}。
%% 不定义 \sym 就是 Undefined control sequence；定义了但不换字体，
%% 则那些字符在 PDF 上留空白且**不报错**（更糟）。
%% 用 Segoe UI Symbol：覆盖这些符号，且是单色字体
%% （彩色 emoji 字体在 XeLaTeX 里会渲染成空白或黑块）。
\newfontfamily\symfont{Segoe UI Symbol}[AutoFakeBold=1.2]
\newcommand{\sym}[1]{{\symfont #1}}

%% ---------- 版面 ----------
\usepackage[a4paper,margin=2.05cm,columnsep=0.72cm]{geometry}
\usepackage{amsmath,amssymb}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{tabularx,array,longtable,multirow}
\usepackage{caption}
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage{xcolor}
\usepackage{fancyhdr}
\usepackage[hidelinks]{hyperref}
\usepackage{xurl}

%% URL 断行：参考文献那一节的关键。
%% 不加这三件，LaTeX 会把 URL 当不可断的长 token，
%% 为了塞下它而把同段的中文标题逐词拆行。
\Urlmuskip=0mu plus 1mu\relax
\makeatletter
\g@addto@macro\UrlBreaks{\do\/\do\.\do\-\do\_\do\:}
\makeatother
\newenvironment{refs}{\par\raggedright\small}{\par}

\usepackage{cleveref}
\usepackage{microtype}
\usepackage{listings}
\usepackage{placeins}
\lstset{
  basicstyle=\ttfamily\scriptsize,
  breaklines=true,
  breakatwhitespace=false,
  columns=fullflexible,
  frame=none,
  xleftmargin=0.6em,
  aboveskip=4pt,
  belowskip=4pt
}

\captionsetup{
  font=small,
  labelfont=bf,
  labelsep=quad,
  skip=4pt,
  justification=centering
}
\captionsetup[table]{position=top}

%% ---------- 中文断行 ----------
%% \XeTeXlinebreaklocale "zh" 让 XeTeX 采用中文禁则，
%% 从而在标点之后造出断行点（ctex 在 zh 下本该做但此处未生效的一步）。
\XeTeXlinebreaklocale "zh"
\XeTeXlinebreakskip = 0pt plus 0.08em minus 0.02em
\XeTeXlinebreakpenalty=0

\setlength{\emergencystretch}{5em}
\tolerance=5000
\hbadness=5000
\hyphenpenalty=50
\relpenalty=50

%% 标题层级
\titleformat{\section}{\large\bfseries\sffamily}{\thesection}{0.6em}{}
\titleformat{\subsection}{\normalsize\bfseries\sffamily}{\thesubsection}{0.5em}{}
\titleformat{\subsubsection}{\normalsize\bfseries}{\thesubsubsection}{0.4em}{}
\titlespacing*{\section}{0pt}{10pt plus 2pt minus 2pt}{5pt plus 1pt}
\titlespacing*{\subsection}{0pt}{7pt plus 1pt minus 1pt}{3pt plus 1pt}
\titlespacing*{\subsubsection}{0pt}{5pt plus 1pt minus 1pt}{2pt}

\setlist[itemize]{topsep=2pt,itemsep=1pt,parsep=0pt,leftmargin=1.4em}
\setlist[enumerate]{topsep=2pt,itemsep=1pt,parsep=0pt,leftmargin=1.6em}
\newenvironment{tightlist}{\begin{itemize}\raggedright}{\end{itemize}}

\linespread{1.28}
\setlength{\parskip}{3.2pt plus 1pt}
\setlength{\parindent}{1.6em}

\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[C]{\small\thepage}

%% 浮动体放宽：双栏默认允许的浮动体太少，容易全部堆到文末。
\renewcommand{\topfraction}{0.92}
\renewcommand{\bottomfraction}{0.85}
\renewcommand{\textfraction}{0.08}
\renewcommand{\floatpagefraction}{0.80}
\setcounter{topnumber}{4}
\setcounter{bottomnumber}{3}
\setcounter{totalnumber}{8}

%% 圈码：Times New Roman 缺 U+2460 区段字形，
%% 编译只报 Missing character 不报错，PDF 上留空白。用 CJK 字体渲染。
\newcommand{\textcjkc}[1]{{\CJKfamily{}#1}}
